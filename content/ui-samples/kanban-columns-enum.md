It is possible to define columns using an enumeration for the Kanban board component. To do this, you must specify the columnsEnum attribute for the <columns/> element. In this way, all values of the enumeration will be associated with columns, and the column header will be automatically set to the localized value of the enumeration. Hierarchical columns are not supported when defined using an enumeration.

<view xmlns="http://jmix.io/schema/flowui/view"

                       height="100%"
                       collapsible="true"
                       taskTagsVisible="true">
            <kanban:propertiesMapping id="id"
                                      status="status"
                                      text="name"
                                      color="color"
                                      tags="tags"/>
            <kanban:columns columnsEnum="io.jmix.uisamples.entity.TaskStatus"/>
        </kanban:kanban>
    </layout>

package io.jmix.uisamples.entity;

import io.jmix.core.metamodel.datatype.EnumClass;
import org.springframework.lang.Nullable;

public enum TaskStatus implements EnumClass<String> {

    TODO("todo"),
    IN_PROGRESS("in_progress"),
    TESTING("testing"),
    TESTING_MANUAL("manual"),
    TESTING_AUTO("auto"),
    DONE("done");

    private final String id;

    TaskStatus(String id) {
        this.id = id;
    }

    @Override
    public String getId() {
        return id;
    }

    @Nullable
    public static TaskStatus fromId(String id) {
        for (TaskStatus at : TaskStatus.values()) {
            if (at.getId().equals(id)) {
                return at;
            }
        }
        return null;
    }
}
